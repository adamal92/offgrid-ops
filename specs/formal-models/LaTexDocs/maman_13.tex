\subsection*{א}

\begin{letyehi}
$A = \{ 1, 2 \}$ קבוצה.

\begin{ntp}
את כל הפונקציות
$ A \rightarrow \Bbb{N} $
שאינן חד חד ערכיות (להלן: חח"ע)
\end{ntp}

\begin{define}
    $ f: A \rightarrow B := (A, B, F ) \; | \; F \subseteq A \times B $
\end{define}

\newline

ניקח לדוגמא את כל הפונקציות
$ F : A \rightarrow M \subseteq \Bbb{N} $
שמקיימות:

\begin{letyehi}
    $ F = \{ (1, m), (2, m) \} \; | \; m \in M \subseteq \Bbb{N} $
\end{letyehi}

אלו הן הפונקציות שאינן חח"ע.

\subsection*{ב}

ניקח לדוגמא את כל הפונקציות
$ F : \Bbb{N} \rightarrow A $
שמקיימות:

\begin{letyehi}
    $ F = \{ (n, 1) \} \; | \; n \in \Bbb{N} $
\end{letyehi}
\begin{letyehi}
    $ F = \{ (n, 2) \} \; | \; n \in \Bbb{N} $
\end{letyehi}

אלו הן הפונקציות שאינן על.

\subsection*{ג}

\begin{letyehi}
    $ f : A \rightarrow \Bbb{N} $
\end{letyehi}

\begin{letyehi}
    $ g : \Bbb{N} \rightarrow A $
\end{letyehi}

\subsubsection*{a.}

\begin{ntp}
    $ \exists (g \circ f)^{-1} $
\end{ntp}

\begin{define}
    $ g \circ f = g(f(x)) $
\end{define}

\begin{proof}
    \begin{sen}
        $ \iff Dom(f) \subseteq Dom(g) \land Img(g) \subseteq Img(f) $
        הפיכה $ f \circ g $
    \end{sen}
    \[\Downarrow\]
    $ \iff Dom(g) \subseteq Dom(f) \land Img(f) \subseteq Img(g) $
    הפיכה $ g \circ f $
    \[
        g \circ f: \Bbb{N} \rightarrow \Bbb{N}
    \]
    \[
        \Bbb{N} \not \subseteq A
    \]
    \[\Downarrow\]
    המקור של $g$
    אינו מוכל במקור של $f$,
    וגם התמונה של $f$
    אינה מוכלת בתמונה של $g$
    ולכן ההרכבה $f$
    היא אינה הפיכה.
    \[\Downarrow\]
    \[
        \lnot \exists (g \circ f)^{-1}
    \]    
\end{proof}

\begin{ntp}
    $ \exists (f \circ g)^{-1} $
\end{ntp}


\begin{proof}
    \begin{sen}

        הפיכה $ f \circ g $
    \end{sen}

        f \circ g: A \rightarrow A
    \]
    \[
        A \subseteq \Bbb{N}
    \]
    \[\Downarrow\]
    המקור של $f$
    מוכל במקור של $g$,

    \[\Downarrow\]

    \]    
\end{proof}

\section*{שאלה 2}

\subsection*{א}

\begin{letyehi}
    $ f: A \rightarrow B , C \subseteq B $
\end{letyehi}

\begin{ntp}
    $ f(f^{-1}(C)) \subseteq C $
\end{ntp}

\begin{proof}
    \begin{define}
        $ A \subseteq B := \{ x \: | \: x \in A \Rightarrow x \in B : \forall x \} $
    \end{define}
    \begin{define}
        $ f^{-1}(C) := \{ x \: | \: f(x) \in C : x \in A \} $
    \end{define}
    \begin{letyehi}
            \[
                y \in f(f^{-1}(C))
            \]
    \end{letyehi}
    \[
        \exists x \in f^{-1}(C) : f(x) = y
    \]
    \[
        f(x) \in C
    \]
    \[ \Downarrow \]
    \[
        \boxed{y \in C}
    \]
    \[ \Downarrow \]
    \[
        y \in f(f^{-1}(C)) \Rightarrow y \in C \hspace{0.5cm} \square
    \]
    \[ \Downarrow \]
    \[
        f(f^{-1}(C)) \subseteq C
    \]
\end{proof}

\subsection*{ב}

\begin{define}
    $ A \subset B := A \subseteq B \land A \not = B = \{ x | x \in A \Rightarrow x \in B \land A \not = B \} $
\end{define}

ניקח לדוגמא את הפונקציה המוגדרת בצורה הבאה:

\begin{ex}
    \begin{letyehi}
        $ A = \{ 1, 2 \} $
    \end{letyehi}
    
    \begin{letyehi}
        $ B = \{ 1, 2, 3, 4 \} $
    \end{letyehi}
    
    \begin{letyehi}
        $ C = \{ 1, 2, 3 \} $
    \end{letyehi}
    קבוצות.

    \begin{letyehi}
        $ f = (A, B, \{ (a, 1), (b, 2) \}) $
    \end{letyehi}
   
    \begin{letyehi}
        $ f^{-1} = (B, A, \{ (1, a), (2, b), (3, a), (4, b) \}) $
    \end{letyehi}
    הפונקציה ההופכית ל $f$.

    \[ \Downarrow \]

    \[
        f(f^{-1}(C)) = (C, \{ 1, 2 \}, \{ (1, 1), (2, 2), (3, 3) \})
    \]

    \[
        \{ 1, 2 \} \subset (\{ 1, 2, 3 \} = C)
    \]

    \[ \Downarrow \]

    מצאנו דוגמא למצב שבו:
    \[
        f(f^{-1}(C)) \subset C
    \]
\end{ex}

\end{ntp}

    \]
    
    \[ \Downarrow \]

        \]

    \end{lem}
    
    \[ \Downarrow \]

        f(f^{-1}(C)) = C \hspace{1cm} \square
    \]
\end{proof}

    $f$ אינה על

\end{define}

\begin{proof}
    \begin{ex}
        \begin{ass}
            $ fx = 2x - 1 $
        \end{ass}
        \begin{ass}
            $ gx = x $
        \end{ass}
    \end{ex}

        (f \circ g) (n) = 2n-1
    \]
    \[
        fx = 2
    \]

    \]

    \]

    \]
    \[\Downarrow\]
    \[
        \boxed{fx \not = 2 | x \in \Bbb{N}}
    \]

מצאנו דוגמא שלפיה הפונקציה היא אינה על. \square

\end{proof}

\begin{letyehi}
    $ f, g : A \rightarrow A $
\end{letyehi}

\subsubsection*{a}

\begin{ntp}
    אם $f$ אינה על אז $ f \circ g $ גם היא אינה על
\end{ntp}

\begin{define}
    $ := \{ f(a) = b \; | \; \exists a \in A : \forall b \in B \} $
    על $f$
\end{define}

\begin{proof}
    \begin{remark}
        $ \Rightarrow \exists b \in A : f(a) \not = b \; | \; \forall a \in A $
        $f$ אינה על
    \end{remark}

    $ \iff (f \circ g)(a) = b \; | \; \exists a \forall b : a, b \in A $
    $ f \circ g $ על

\begin{letyehi}
    $ g(a) = j $
\end{letyehi}

\[\Downarrow\]

\[
    f \circ g _{onto} \iff f(j) = b \; | \; \exists a \forall b : a, b \in A
\] על
\[
    \exists b : f(a) \not = b \; | \; \forall a \in A
\]
\[\Downarrow\]
\[
    \exists b : f(j) \not = b \; | \; \forall a \in A
\]
\[\Downarrow\]
\[
    \exists b : f(g(a)) \not = b \; | \; \forall a \in A
\]
\[\Downarrow\]
\[
    \boxed{\lnot \exists b : f(g(a)) = b \; | \; \forall b \in A}
\]
\[\Downarrow\]
$ f \circ g $ אינה על
\square

\end{proof}

\subsection*{ב}

\begin{letyehi}
    $ f, g : \Bbb{N} \rightarrow \Bbb{N} $
\end{letyehi}

\begin{remark}
    $ g(n) := 2n - 1 \; | \; \forall n \in \Bbb{N} $
\end{remark}

\begin{remark}
    \begin{align*}
        f(n) :=
        \begin{cases}
            \frac{n+1}{2} \\
            \; \; 1
        \end{cases}
        \Bigg | \; \forall n \in \mathbb{N}
    \end{align*}
\end{remark}

\begin{define}
    $ := \{ f(a) = b \: | \: \exists a \in A : \forall b \in B \} $
    על $f$
\end{define}

\begin{proof}
    
    \begin{ass}
        בשלילה
        $ \Rightarrow g(a) = b \: | \: \exists a \forall b : a, b \in A $
        $g$ על
    \end{ass}
    
    \[\Downarrow\]
    
    \[
        g(n) = b \; | \; \exists n \forall b : n, b \in \Bbb{N}
    \]
    \[
        2n - 1 = b
    \]
    \[
        n = \frac{b+1}{2}
    \]
    
    \begin{ex}
        נגדית
        $ b=2 $
    \end{ex}
    
    \[\Downarrow\]
    
    \[
        \boxed{ n = \frac{2+1}{2} = \frac{3}{2} \not \in \Bbb{N} }
    \]
    \[\Downarrow\]
    \[
        \exists b : g(n) \not = b \; | \; \forall n \in \Bbb{N} : b \in \Bbb{N}
    \]
    \[\Downarrow\]
    $ g $ אינה על \square

    \begin{align*}
        f \circ g = f(g(n)) = 
        \begin{cases}
            \frac{(2n - 1) + 1}{2} = \frac{2n}{2} \\
            \; \; 1
        \end{cases}
        =
        \begin{cases}
            n \\
            1
        \end{cases}
    \end{align*}
    \[
        g(n) = 2n - 1
    \]
    \[\Downarrow\]
    \[
        g(n) : \Bbb{N} \rightarrow odd
    \]
    \[\Downarrow\]
    \[
        g(n) \in odd
    \]
    \[\Downarrow\]
    \[
        \boxed{ f(g(n)) = n } \; | \; \forall n \in \Bbb{N} : \exists n
    \]
    \[\Downarrow\]
    $f \circ g$ על. \square

\end{proof}
